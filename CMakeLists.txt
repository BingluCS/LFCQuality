cmake_minimum_required(VERSION 3.18)
project(LFCQuality VERSION 0.2.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(LFCQUALITY_OPENMP "Use OpenMP kernels" ON)
option(LFCQUALITY_FFTW "Enable FFTW spectra" ON)
add_library(lfcquality SHARED src/lfcquality/native/kernels.cpp src/lfcquality/native/api.cpp)
target_compile_features(lfcquality PRIVATE cxx_std_17)
target_include_directories(lfcquality PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/lfcquality/native> $<INSTALL_INTERFACE:include> $<INSTALL_INTERFACE:include/lfcquality>)
target_compile_options(lfcquality PRIVATE -O3 -fno-fast-math -ffp-contract=off)
set_target_properties(lfcquality PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
add_library(LFCQuality::lfcquality ALIAS lfcquality)
if(LFCQUALITY_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(lfcquality PRIVATE OpenMP::OpenMP_CXX)
endif()
if(LFCQUALITY_FFTW)
  find_path(FFTW_INCLUDE_DIR fftw3.h REQUIRED)
  find_library(FFTW_LIBRARY fftw3 REQUIRED)
  find_library(FFTW_THREADS_LIBRARY fftw3_threads REQUIRED)
  find_package(Threads REQUIRED)
  target_include_directories(lfcquality PRIVATE ${FFTW_INCLUDE_DIR})
  target_link_libraries(lfcquality PRIVATE ${FFTW_THREADS_LIBRARY} ${FFTW_LIBRARY} Threads::Threads)
  target_compile_definitions(lfcquality PRIVATE LFQ_FFTW LFQ_FFTW_THREADS)
endif()
add_executable(lfcquality_example examples/native_example.cpp)
target_link_libraries(lfcquality_example PRIVATE lfcquality)
add_executable(lfcquality_c_example examples/c_api.c)
target_compile_features(lfcquality_c_example PRIVATE c_std_11)
target_link_libraries(lfcquality_c_example PRIVATE lfcquality)
add_executable(lfcquality_cpp_example examples/cpp_api.cpp)
target_compile_features(lfcquality_cpp_example PRIVATE cxx_std_17)
target_link_libraries(lfcquality_cpp_example PRIVATE lfcquality)
add_executable(lfcquality_native_api_test tests/native_api_test.c)
target_compile_features(lfcquality_native_api_test PRIVATE c_std_11)
target_link_libraries(lfcquality_native_api_test PRIVATE lfcquality)
add_executable(lfcquality_cpp_api_test tests/native_cpp_test.cpp)
target_compile_features(lfcquality_cpp_api_test PRIVATE cxx_std_17)
target_link_libraries(lfcquality_cpp_api_test PRIVATE lfcquality)
if(UNIX)
  target_link_libraries(lfcquality_c_example PRIVATE m)
  target_link_libraries(lfcquality_native_api_test PRIVATE m)
endif()
include(CTest)
add_test(NAME native_example COMMAND lfcquality_example)
add_test(NAME c_api_example COMMAND lfcquality_c_example)
add_test(NAME cpp_api_example COMMAND lfcquality_cpp_example)
add_test(NAME c_api_edge_cases COMMAND lfcquality_native_api_test)
add_test(NAME cpp_api_lifetime COMMAND lfcquality_cpp_api_test)
set(lfcquality_cmake_dir "${CMAKE_INSTALL_LIBDIR}/cmake/LFCQuality")
install(TARGETS lfcquality EXPORT LFCQualityTargets
        LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
        ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
        RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES src/lfcquality/native/kernels.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/lfcquality")
install(DIRECTORY src/lfcquality/native/lfcquality/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/lfcquality"
        FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp")
configure_package_config_file(cmake/LFCQualityConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/LFCQualityConfig.cmake"
                              INSTALL_DESTINATION "${lfcquality_cmake_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/LFCQualityConfigVersion.cmake"
                                 VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
install(EXPORT LFCQualityTargets NAMESPACE LFCQuality:: DESTINATION "${lfcquality_cmake_dir}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/LFCQualityConfig.cmake"
              "${CMAKE_CURRENT_BINARY_DIR}/LFCQualityConfigVersion.cmake" DESTINATION "${lfcquality_cmake_dir}")
